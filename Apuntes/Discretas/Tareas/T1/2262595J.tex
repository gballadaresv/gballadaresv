\section*{Pregunta 1}

\subsection*{(a)}

Por demostrar que:
\begin{equation}
    \label{eq:1}
    \sum_{i=0}^{n} 2^i = 2^{n+1} - 1
\end{equation}

\begin{itemize}
    \item \textbf{CB:}
    Para $n=0$ se cumple que
        \begin{equation*}
            \sum_{i=0}^{0} 2^i = 2^0 = 1 = 2^{0+1} - 1
        \end{equation*}

    \item \textbf{HI:}
    Supondremos que (\ref*{eq:1}) se cumple para un $n \in \N$

    \item \textbf{TI:}
    Queremos demostrar por inducción que (\ref*{eq:1}) se cumple también para $n + 1$

    \begin{align*}
        \sum_{i=0}^{n+1} 2^i
        &= 2^{n+1} + \sum_{i=0}^{n} 2^i \\
        &\stackrel{HI}{=} 2^{n+1} + 2^{n+1} - 1 \\
        &= 2(2^{n+1}) - 1 \\
        &= 2^{n+2} - 1 \\
        &= 2^{(n+1)+1} - 1
    \end{align*}
\end{itemize}

Con lo anterior podemos concluir por inducción simple que la propiedad (\ref*{eq:1}) se cumple para todos los números naturales.

\subsection*{(b)}
Por demostrar que:
\begin{equation}
    \label{eq:2}
    (m + 1)^n > mn
\end{equation}

\begin{itemize}
    \item \textbf{CB:}
    Notamos que para $n = 1$ se cumple que
    \begin{equation*}
        (m + 1)^1 > m \cdot 1
    \end{equation*}

    \item \textbf{HI:}
    Supondremos que (\ref*{eq:2}) se cumple para  un $n \geq 1$ cualquiera.

    \item \textbf{TI:}
    Queremos demostrar por inducción que (\ref*{eq:2}) se cumple también para $n+1$
    \begin{align*}
        (m + 1)^{n+1}
        &= (m + 1)^n \cdot (m + 1)
    \end{align*}
    Nótese que el término $(m + 1)^{n+1}$ puede acotarse inferiormente según la hipótesis inductiva,
    resultando en lo siguiente:
    \begin{align*}
        (m + 1)^n \cdot (m + 1) > mn(m + 1) = m^2n + mn
    \end{align*}
    Finalmente, podemos utilizar la siguiente relación para acotar nuevamente 
    la expresión $m^2n + mn$.
    \begin{align*}
        m^2n + mn \geq mn + m = m(n+1)
    \end{align*}
    Resumiendo, tenemos que:
    \begin{align*}
        (m + 1)^{n+1} > m(n+1)
    \end{align*}
    Se demostró entonces por inducción simple que la propiedad \ref*{eq:2} se cumple 
    para todo número natural $m \geq 1$ 
\end{itemize}

\section*{Pregunta 2}
Por demostrar que todo número natural $n \geq 0$ se puede escribir en base $b$ según:
\begin{equation}
    \label{eq:3}
    n = \sum_{i=0}^{\ell-1} k_i \cdot b^i
\end{equation}
\begin{itemize}
    \item \textbf{CB:}
    Considerando el caso $n = 0$ se cumple que:
    \begin{equation*}
        0 = \sum_{i=0}^{1-1} k_i \cdot b^i = 0 \cdot b^0
    \end{equation*}
    Sea ahora $r$ un número natural de largo $\ell = 1$ tal que $r < b$:
    \begin{equation*}
        r = \sum_{i=0}^{1-1} k_i \cdot b = r \cdot b^0
    \end{equation*}

    \item \textbf{HI:}
    Sea $j$ un número natural tal que $j \leq n$, supondremos que (\ref*{eq:3}) se cumple
    para $j$.

    \item \textbf{TI:}
    Sabemos que un número $n$ en base $b$ puede ser escrito de la forma
    $n = j \cdot b + r$. Con lo anterior, y utilizando la hipótesis de indicción, se obtiene que:
    \begin{align*}
        n+1 &= j \cdot b + r \\
        &\stackrel{HI}{=} (\sum_{i=0}^{\ell-1} k_i \cdot b^i) \cdot b + r \\
        &= \sum_{i=0}^{\ell-1} k_i \cdot b^{i+1} + r \\
        &\stackrel{CB}{=} \sum_{i=0}^{\ell-1} k_i \cdot b^{i+1} + rb^0
    \end{align*}
    Si renombramos las variables de la sumatoria de forma conveniente, siendo cada $k_i$ denominada ahora
    como $h_{i+1}$, y renombrando $r$ a $h_0$, se puede reescribir la expresión anterior como:
    \begin{equation*}
        n + 1 = \sum_{i=0}^{\ell-1} h_i \cdot b^{i+1}
    \end{equation*}
    Con lo anterior, se concluye que todo número natural $n \geq 0$ se puede representar
    en base $b \geq 2$.
\end{itemize}
