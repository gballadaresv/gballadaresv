\section*{Pregunta 1}

\subsection*{(a)}

Sean $a \in \Sigma$ y  $w \in \Sigma^*$. Definimos la función $\#_{a} : \Sigma^* \ra \N$ 
de la siguiente manera:
\begin{enumerate}
    \item $\#_{a}(\varepsilon ) = 0$
    \item Si $\#_{a}(wa) = n$, entonces $\#_{a}(wa) = n+1$
\end{enumerate}

\subsection*{(b)}

Sean $a, b, t \in \Sigma$ y $w \in \Sigma^*$. Definimos la función 
$r_{a \to b}: \Sigma^* \ra \Sigma^*$ de la siguiente manera:

\begin{enumerate}
    \item $r_{a \to b}(\varepsilon) = \varepsilon$
    \item Si $r_{a \to b}(w) = w$:
    \begin{enumerate}
        \item $t \neq a: \\
        r_{a \to b}(wt) = wt$
        \item $t = a: \\
        r_{a \to b}(wt) = wb$
    \end{enumerate}
\end{enumerate}
